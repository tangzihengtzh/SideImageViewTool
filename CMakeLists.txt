cmake_minimum_required(VERSION 3.21)
project(ScreenDock VERSION 1.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
add_executable(ScreenDock WIN32 src/main.cpp src/core.cpp src/history_index.cpp src/ui.cpp src/app.cpp src/dock.cpp src/pin.cpp src/capture.cpp tests/self_test.cpp src/app.manifest "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_compile_definitions(ScreenDock PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 SCREENDOCK_VERSION=L"${PROJECT_VERSION}")
target_compile_options(ScreenDock PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(ScreenDock PRIVATE d2d1 dwrite windowscodecs dwmapi comctl32 shell32 ole32 user32 gdi32 shlwapi)
target_include_directories(ScreenDock PRIVATE src)
set_target_properties(ScreenDock PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
enable_testing()
add_test(NAME ScreenDockSelfTest COMMAND ScreenDock --self-test "${CMAKE_SOURCE_DIR}/tests/output")
